Functions & Macros

import console

fn factorial(n: int): int
    if n <= 1
        return 1
    return n * factorial(n - 1)

let PI be 3.14159

void main()
    print("5! = ", factorial(5))
    print("Circumference of circle: ", 2.0 * PI * 5.0)

Classes & Inheritance

import console

class Animal
    name: string
    age: int = 0

    void speak()
        print(this.name, " makes a sound")

class Dog : Animal
    breed: string = "Mixed"

    void speak()
        print(this.name, " barks!")

class Cat : Animal
    void speak()
        print(this.name, " meows!")

void main()
    let dog = Dog(name: "Rex", age: 3, breed: "Labrador")
    let cat = Cat(name: "Whiskers", age: 5)

    dog.speak()
    cat.speak()

Generics

import console

fn first(arr: [T]): T
    return arr[0]

fn last(arr: [T]): T
    return arr[#arr - 1]

fn reverse(arr: [T])
    let i = 0
    let j = #arr - 1
    while i < j
        let tmp = arr[i]
        arr[i] = arr[j]
        arr[j] = tmp
        i = i + 1
        j = j - 1

void main()
    let ints = [1, 2, 3, 4, 5]
    let strs = ["a", "b", "c"]

    print("First int: ", first(ints))
    print("Last int: ", last(ints))
    print("First string: ", first(strs))

    reverse(ints)
    print("Reversed: ", ints)

Control Flow

import console

void main()
    let count = 0
    while count < 5
        if count % 2 == 0
            print(count, " is even")
        else
            print(count, " is odd")
        count = count + 1

    for i in 0 to 3
        print("Loop: ", i)

    let x = 10
    if x > 5 and x < 15
        print("x is between 5 and 15")
