Standard Library

Console (I/O)

Print output and read input:

import console

// Output
print("Hello")
print("Value: ", 42, true)  // multiple arguments

// Input (reads a line)
let line = input()
let char = input_char()

Math

Mathematical functions:

import math

// Basic operations
let abs_val = math::abs(-42)
let max_val = math::max(10, 20)
let min_val = math::min(10, 20)

// Trigonometry
let s = math::sin(3.14159 / 2.0)
let c = math::cos(0.0)

// Power and root
let squared = math::pow(2.0, 3.0)    // 2^3
let root = math::sqrt(16.0)

// Rounding
let rounded = math::round(3.7)
let floored = math::floor(3.7)
let ceiled = math::ceil(3.2)

String Methods

Strings have built-in methods:

let s = "hello world"

// Length
let len = #s        // 11

// Character access
let c = s[0]        // 'h'

// Substring
let sub = s[0 to 5] // "hello"

// Conversion
let num = s.to_int()    // parse to int
let str = (42).to_string()

Array Methods

Arrays have built-in operations:

let arr = [3, 1, 4, 1, 5]

// Length
let len = #arr      // 5

// Access
let first = arr[0]  // 3
arr[1] = 10

// Slice
let slice = arr[0 to 3] // [3, 10, 4]

// Convert to string
let str = arr.to_string()

Algorithm Module

Common algorithms for arrays:

import algorithm

let arr = [5, 2, 8, 1, 9]

// Sort (modifies in place)
algorithm::sort(arr)

// Search
let idx = algorithm::find(arr, 8)

// Min/Max
let min_val = algorithm::min(arr)
let max_val = algorithm::max(arr)

Type Conversion

Built-in conversion methods on types:

// To string
let s1 = (42).to_string()
let s2 = (3.14).to_string()

// From string
let i = "42".to_int()        // int?
let f = "3.14".to_float()    // float?

// Type casting
let f = 42 as float
let i = 3.14 as int

Building Standard Library

The standard library is implemented in C in stdlib/. It's automatically linked into every x86 build.

stdlib/
  runtime.c         # core runtime (memory, operator lowerings)
  console.c         # I/O functions
  math.c            # mathematical functions
  algorithm.c       # array algorithms
  km_runtime.h      # header definitions
